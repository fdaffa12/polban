class DepartmentController extends Controller

    private function hasContentManagementAccess($user)
    {
        return in_array($user->role, ["BPH", "MEDKOM"]);
    }
    
    public function index()
    {
        $user = Auth::user();

        // Check if user has access to departments
        if (!$this->hasContentManagementAccess($user)) {
            abort(403, 'Unauthorized action.');
        }
        
        return Inertia::render('Departments/Index', [
            'departments' => Department::with('members')->latest()->get(),
            'deptLogos' => DeptLogo::latest()->get(),
        ]);
    }

    public function storeDepartment(Request $request)
    {
        $request->validate([
            'dept_name' => 'required|string|max:255',
            'image' => 'required|image|max:10248',
            'desc' => 'nullable|string|max:1000',
        ]);

        try {
            $image = $request->file('image');
            $imageName = time() . '.' . $image->getClientOriginalExtension();

            // Path FISIK untuk server
            $uploadPath = '/home/u928398125/domains/hmjtk-polban.com/public_html/storage/departments';

            // Buat folder jika belum ada
            if (!File::exists($uploadPath)) {
                File::makeDirectory($uploadPath, 0777, true);
            }

            // Simpan file
            $image->move($uploadPath, $imageName);

            // Path RELATIF untuk database
            $path = 'departments/' . $imageName;

            Department::create([
                'dept_name' => $request->dept_name,
                'image' => $path,
                'desc' => $request->desc,
            ]);

            return redirect()->back()->with('success', 'Department added successfully');
        } catch (\Exception $e) {
            Log::error('Department creation error: ' . $e->getMessage());
            return redirect()->back()->with('error', 'Failed to add department');
        }
    }

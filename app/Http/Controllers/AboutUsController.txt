<?php

namespace App\Http\Controllers;

use App\Models\AboutUs;
use Illuminate\Http\Request;
use Inertia\Inertia;
use Illuminate\Support\Facades\File;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\Auth;

class AboutUsController extends Controller
{
    private function hasContentManagementAccess($user)
    {
        return in_array($user->role, ["BPH", "MEDKOM"]);
    }

    public function index()
    {
        $user = Auth::user();

        // Check if user has access to about us
        if (!$this->hasContentManagementAccess($user)) {
            abort(403, 'Unauthorized action.');
        }
        
        $aboutUs = AboutUs::first() ?? new AboutUs();
        return Inertia::render('AboutUs/Index', [
            'aboutUs' => $aboutUs
        ]);
    }

    public function update(Request $request)
    {
        try {
            $request->validate([
                'au_title' => 'required|string|max:255',
                'au_desc' => 'required|string',
                'au_image' => 'nullable|image|max:2048',
                'au_values' => 'nullable',
                'history' => 'nullable',
            ]);

            $aboutUs = AboutUs::first() ?? new AboutUs();
            $data = $request->except(['au_image']);

            // Handle main image
            if ($request->hasFile('au_image')) {
                // Hapus file lama jika ada
                if ($aboutUs->au_image) {
                    $oldPath = '/home/u928398125/domains/iftekno.com/public_html/hmjtk/storage/' . $aboutUs->au_image;
                    if (File::exists($oldPath)) {
                        File::delete($oldPath);
                    }
                }

                $image = $request->file('au_image');
                $imageName = time() . '.' . $image->getClientOriginalExtension();

                // Path FISIK untuk server
                $uploadPath = '/home/u928398125/domains/iftekno.com/public_html/hmjtk/storage/about-us';

                // Buat folder jika belum ada
                if (!File::exists($uploadPath)) {
                    File::makeDirectory($uploadPath, 0777, true);
                }

                // Simpan file
                $image->move($uploadPath, $imageName);

                // Path RELATIF untuk database
                $data['au_image'] = 'about-us/' . $imageName;
            }

            $aboutUs->fill($data);
            $aboutUs->save();

            return redirect()->back()->with('success', 'About Us updated successfully.');
        } catch (\Exception $e) {
            Log::error('About Us update error: ' . $e->getMessage());
            return redirect()->back()->with('error', 'Failed to update About Us');
        }
    }

    public function addImage(Request $request)
    {
        try {
            $request->validate([
                'image' => 'required|image|max:2048',
                'title' => 'required|string|max:255'
            ]);

            $aboutUs = AboutUs::first();
            if (!$aboutUs) {
                return response()->json(['message' => 'About Us not found'], 404);
            }

            $image = $request->file('image');
            $imageName = time() . '_' . uniqid() . '.' . $image->getClientOriginalExtension();

            // Path FISIK untuk server
            $uploadPath = '/home/u928398125/domains/iftekno.com/public_html/hmjtk/storage/about-us/gallery';

            // Buat folder jika belum ada
            if (!File::exists($uploadPath)) {
                File::makeDirectory($uploadPath, 0777, true);
            }

            // Simpan file
            $image->move($uploadPath, $imageName);

            // Path RELATIF untuk database
            $path = 'about-us/gallery/' . $imageName;
            
            $images = $aboutUs->au_multiple_image ?? [];
            $images[] = [
                'path' => $path,
                'title' => $request->title
            ];
            
            $aboutUs->au_multiple_image = $images;
            $aboutUs->save();

            return redirect()->back()->with([
                'success' => true,
                'message' => 'Image added successfully',
                'images' => $aboutUs->au_multiple_image
            ]);
        } catch (\Exception $e) {
            Log::error('About Us add image error: ' . $e->getMessage());
            return response()->json(['message' => 'Failed to add image'], 500);
        }
    }

    public function updateImage(Request $request, $index)
    {
        try {
            $request->validate([
                'title' => 'required|string|max:255',
                'image' => 'nullable|image|max:2048'
            ]);

            $aboutUs = AboutUs::first();
            if (!$aboutUs) {
                return response()->json(['message' => 'About Us not found'], 404);
            }

            $images = $aboutUs->au_multiple_image;
            if (!isset($images[$index])) {
                return response()->json(['message' => 'Image not found'], 404);
            }

            // Update image if new file is uploaded
            if ($request->hasFile('image')) {
                // Delete old image
                $oldPath = '/home/u928398125/domains/iftekno.com/public_html/hmjtk/storage/' . $images[$index]['path'];
                if (File::exists($oldPath)) {
                    File::delete($oldPath);
                }

                $image = $request->file('image');
                $imageName = time() . '_' . uniqid() . '.' . $image->getClientOriginalExtension();

                // Path FISIK untuk server
                $uploadPath = '/home/u928398125/domains/iftekno.com/public_html/hmjtk/storage/about-us/gallery';

                // Buat folder jika belum ada
                if (!File::exists($uploadPath)) {
                    File::makeDirectory($uploadPath, 0777, true);
                }

                // Simpan file
                $image->move($uploadPath, $imageName);

                // Update path
                $images[$index]['path'] = 'about-us/gallery/' . $imageName;
            }

            // Update title
            $images[$index]['title'] = $request->title;
            
            $aboutUs->au_multiple_image = $images;
            $aboutUs->save();

            return redirect()->back()->with([
                'success' => true,
                'message' => 'Image updated successfully',
                'images' => $aboutUs->au_multiple_image
            ]);
        } catch (\Exception $e) {
            Log::error('About Us update image error: ' . $e->getMessage());
            return response()->json(['message' => 'Failed to update image'], 500);
        }
    }

    public function removeImage(Request $request)
    {
        try {
            $request->validate([
                'index' => 'required|integer'
            ]);

            $aboutUs = AboutUs::first();
            if (!$aboutUs) {
                return response()->json(['message' => 'About Us not found'], 404);
            }

            $images = $aboutUs->au_multiple_image ?? [];
            if (!isset($images[$request->index])) {
                return response()->json(['message' => 'Image not found'], 404);
            }

            // Hapus file dari storage
            $imagePath = '/home/u928398125/domains/iftekno.com/public_html/hmjtk/storage/' . $images[$request->index]['path'];
            if (File::exists($imagePath)) {
                File::delete($imagePath);
            }

            // Hapus dari array dan reindex
            array_splice($images, $request->index, 1);
            $aboutUs->au_multiple_image = array_values($images);
            $aboutUs->save();

            return redirect()->back()->with([
                'success' => true,
                'images' => $aboutUs->au_multiple_image
            ]);
        } catch (\Exception $e) {
            Log::error('About Us remove image error: ' . $e->getMessage());
            return response()->json(['message' => 'Failed to remove image'], 500);
        }
    }
}

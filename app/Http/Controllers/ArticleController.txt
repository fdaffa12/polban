<?php

namespace App\Http\Controllers;

use App\Models\Article;
use App\Models\Category;
use Illuminate\Http\Request;
use Inertia\Inertia;
use App\Models\Tag;
use Illuminate\Support\Facades\File;
use Illuminate\Support\Str;
use Illuminate\Support\Facades\Auth;

class ArticleController extends Controller
{
    private function generateUniqueSlug($title, $id = null)
    {
        $slug = Str::slug($title);
        $count = 1;

        $query = Article::where('slug', $slug);
        if ($id) {
            $query->where('id', '!=', $id);
        }

        while ($query->exists()) {
            $slug = Str::slug($title) . '-' . $count;
            $query = Article::where('slug', $slug);
            if ($id) {
                $query->where('id', '!=', $id);
            }
            $count++;
        }

        return $slug;
    }

    private function hasContentManagementAccess($user)
    {
        return in_array($user->role, ["BPH", "MEDKOM"]);
    }


    public function index()
    {

        $user = Auth::user();

        // Check if user has access to articles
        if (!$this->hasContentManagementAccess($user)) {
            abort(403, 'Unauthorized action.');
        }
        return Inertia::render('Articles/Index', [
            'articles' => Article::with(['category', 'tags'])->latest()->get(),
            'categories' => Category::all(),
            'tags' => Tag::all()
        ]);
    }

    public function storeCategory(Request $request)
    {
        $request->validate([
            'name' => 'required|string|max:255'
        ]);

        Category::create($request->all());

        return redirect()->back();
    }

    public function updateCategory(Request $request, Category $category)
    {
        $request->validate([
            'name' => 'required|string|max:255'
        ]);

        $category->update($request->all());

        return redirect()->back();
    }

    public function deleteCategory(Category $category)
    {
        $category->delete();
        return redirect()->back();
    }

    public function store(Request $request)
    {
        $request->validate([
            'title' => 'required|string|max:255',
            'content' => 'required|string',
            'category_id' => 'required|exists:categories,id',
            'featured_image' => 'nullable|image|max:2048',
            'tags' => 'array',
            'status' => 'required|in:draft,publish,highlight'
        ]);

        $data = $request->except('featured_image', 'tags');
        $data['slug'] = $this->generateUniqueSlug($request->title);

        if (!isset($data['status'])) {
            $data['status'] = 'publish';
        }

        if ($request->hasFile('featured_image')) {
            $image = $request->file('featured_image');
            $imageName = time() . '.' . $image->getClientOriginalExtension();

            // Path FISIK untuk server
            $uploadPath = '/home/u928398125/domains/hmjtk-polban.com/public_html/storage/articles';

            // Buat folder jika belum ada
            if (!File::exists($uploadPath)) {
                File::makeDirectory($uploadPath, 0777, true);
            }

            // Simpan file
            $image->move($uploadPath, $imageName);

            // Path RELATIF untuk database
            $data['featured_image'] = 'articles/' . $imageName;
        }

        $article = $request->user()->articles()->create($data);

        if ($request->has('tags')) {
            $tags = collect($request->tags)->map(function ($tagName) {
                return Tag::firstOrCreate(['name' => $tagName, 'slug' => Str::slug($tagName)]);
            });

            $article->tags()->sync($tags->pluck('id'));
        }

        return redirect()->back();
    }

    public function update(Request $request, Article $article)
    {
        $request->validate([
            'title' => 'required|string|max:255',
            'content' => 'required|string',
            'category_id' => 'required|exists:categories,id',
            'featured_image' => 'nullable|image|max:2048',
            'tags' => 'required|json',
            'status' => 'required|in:draft,publish,highlight'
        ]);

        $data = $request->except(['featured_image', 'tags']);

        if ($request->hasFile('featured_image')) {
            // Hapus file lama jika ada
            if ($article->featured_image) {
                $oldPath = '/home/u928398125/domains/hmjtk-polban.com/public_html/storage/' . $article->featured_image;
                if (File::exists($oldPath)) {
                    File::delete($oldPath);
                }
            }

            $image = $request->file('featured_image');
            $imageName = time() . '.' . $image->getClientOriginalExtension();

            // Path FISIK untuk server
            $uploadPath = '/home/u928398125/domains/hmjtk-polban.com/public_html/storage/articles';

            // Buat folder jika belum ada
            if (!File::exists($uploadPath)) {
                File::makeDirectory($uploadPath, 0777, true);
            }

            // Simpan file
            $image->move($uploadPath, $imageName);

            // Path RELATIF untuk database
            $data['featured_image'] = 'articles/' . $imageName;
        }

        // Update slug if title changed
        if ($article->title !== $request->title) {
            $data['slug'] = $this->generateUniqueSlug($request->title, $article->id);
        }

        $article->update($data);

        // Handle tags
        $tagsArray = json_decode($request->tags, true);
        if (is_array($tagsArray)) {
            $tags = collect($tagsArray)->map(function ($tagName) {
                return Tag::firstOrCreate([
                    'name' => $tagName,
                    'slug' => Str::slug($tagName)
                ]);
            });

            $article->tags()->sync($tags->pluck('id'));
        }

        return back()->with('success', 'Article updated successfully');
    }

    public function destroy(Article $article)
    {
        // Hapus file gambar jika ada
        if ($article->featured_image) {
            $imagePath = '/home/u928398125/domains/hmjtk-polban.com/public_html/storage/' . $article->featured_image;
            if (File::exists($imagePath)) {
                File::delete($imagePath);
            }
        }

        $article->delete();
        return redirect()->back();
    }

    public function show(Article $article)
    {
        return Inertia::render('Articles/Show', [
            'article' => $article->load(['category', 'tags', 'user'])
        ]);
    }
}
